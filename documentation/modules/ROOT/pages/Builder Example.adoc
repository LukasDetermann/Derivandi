.This builder would be generated
[%linenums,java]
----
package com.derivandi.builder;

import builder.com.derivandi.Customer;

class CustomerDerivandiBuilder {
   private String name;
   public CustomerDerivandiBuilder withName(String name) {
      this.name = name;
      return this;
   }
   public Customer build() {
      Customer customer = new Customer();
      customer.setName(name);
      return customer;
   }

}
